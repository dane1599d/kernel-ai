## system
type Policy = "interactive" | "round-robin"
cpus() -> int  // CPUs online
memory() -> #{free_mib: int, total_mib: int, heap_kib: int}  // MiB; kernel heap in KiB
uptime_ms() -> int  // milliseconds since boot
processes() -> [#{pid: int, name: string}]  // the running services
threads() -> #{switches: int, ejections: int, migrations: int}  // scheduler counters since boot
get_policy() -> Policy  // the scheduling policy
set_policy(name: Policy) [undo]  // every CPU
process_stop(name: string) [ask]  // stops or kills processes by name
service_stop|start|restart(name: string) [ask|undo|ask]  // start runs a stopped one; stop keeps it down; restart resets
tune(target: string?, intent: string?) [ask]  // proposes a scheduler trial for a process
tune_undo() [undo]  // puts back the policy a kept trial replaced
